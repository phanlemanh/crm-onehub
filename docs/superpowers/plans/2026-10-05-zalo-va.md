> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Zalo tạm lỗi (408, 429, 5xx, mạng, sai khuôn) không còn bị báo «token sai» hay làm xoá bot; câu chào của bot trỏ đúng trang «Nhắc buổi sáng»; thước ngày cuối kỳ và năng lực Zalo báo đúng; mỗi lần `getMe` lỗi để lại một hàng đếm được.

**Tech Stack:** Bun 1.3.12, TypeScript, Prisma (Postgres), zod, eve (agent). Zalo giả ở `apps/agent/test/support/zalo-gia.ts`.

**Spec:** `docs/superpowers/specs/2026-10-05-zalo-va-design.md` · hợp đồng `_acceptance/zalo-va/contract.md` · evals `_acceptance/zalo-va/evals.yaml`.

```ts
import { createHash } from "node:crypto";

export const ZALO_GETME = {
	loai: "zalo.getme-loi",
	phien: "zalo-bot",
	vanTayKyTu: 12,
} as const;

export function vanTay(token: string): string {
	return createHash("sha256")
		.update(token)
		.digest("hex")
		.slice(0, ZALO_GETME.vanTayKyTu);
}
```

```ts
// apps/agent/agent/lib/zalo-getme.ts
import { randomUUID } from "node:crypto";
import { db } from "@crm/db";
import { vanTay, ZALO_GETME } from "@crm/db/zalo-getme";
import { maLoiZalo, parseZaloGetMe } from "@crm/validation/zalo-update";
import { ZALO } from "./zalo-config";

export type KetLuanGetMe = "tam-loi" | "token-sai";
export type NoiGetMe = "dat-bot" | "kiem-bot";
export type LoiGetMe = {
	lop: string;
	http: number | null;
	maThan: number | null;
	ketLuan: KetLuanGetMe;
};
export type KetGetMe =
	| { loai: "duoc"; tenBot: string }
	| { loai: "loi"; loi: LoiGetMe };
export type PhanGetMe = { status: number; than: unknown };
export type GhiGetMe = (noi: NoiGetMe, token: string, loi: LoiGetMe) => Promise<void>;

function laTam(ma: number | null): boolean {
	if (ma === null) return false;
	return ZALO.http.tamLoi.some((m) => m === ma) || ma >= ZALO.http.loiMayChu;
}

function laTuChoi(ma: number | null): boolean {
	return ma !== null && ma >= ZALO.http.tuChoi.tu && ma < ZALO.http.tuChoi.den;
}

function loi(lop: string, http: number | null, maThan: number | null, tuChoi: boolean): KetGetMe {
	return { loai: "loi", loi: { lop, http, maThan, ketLuan: tuChoi ? "token-sai" : "tam-loi" } };
}

export function phanLopGetMe(phan: PhanGetMe | null): KetGetMe {
	if (phan === null) return loi("mang", null, null, false);
	const maThan = maLoiZalo(phan.than);
	if (phan.status !== ZALO.http.ok) {
		const tuChoi = laTuChoi(phan.status) && !laTam(phan.status) && !laTam(maThan);
		return loi(`http-${phan.status}`, phan.status, maThan, tuChoi);
	}
	if (maThan !== null) {
		return loi(`ok-false-${maThan}`, phan.status, maThan, laTuChoi(maThan) && !laTam(maThan));
	}
	try {
		return { loai: "duoc", tenBot: parseZaloGetMe(phan.than).tenBot };
	} catch {
		return loi("sai-khuon", phan.status, null, false);
	}
}

export const ghiGetMeLoi: GhiGetMe = async (noi, token, l) => {
	await db.agentEvent.create({
		data: {
			id: randomUUID(),
			sessionId: ZALO_GETME.phien,
			type: ZALO_GETME.loai,
			data: { noi, ...l, vanTay: vanTay(token) },
			emittedAt: new Date(),
		},
	});
};

export async function ghiAnToan(ghi: GhiGetMe, noi: NoiGetMe, token: string, l: LoiGetMe): Promise<void> {
	try {
		await ghi(noi, token, l);
	} catch (e) {
		console.error(`[zalo] ghi su kien getMe hong: ${e instanceof Error ? e.message : String(e)}`);
	}
}
```

```ts
async function docBot(token: string, fetcher: typeof fetch, now: Date, ghi: GhiGetMe): Promise<boolean> {
	let phan: PhanGetMe | null = null;
	try {
		phan = await zaloGoiTho(fetcher, token, "getMe", {});
	} catch (loi) {
		console.error(`[zalo] getMe loi mang: ${String(loi)}`);
	}
	const ket = phanLopGetMe(phan);
	if (ket.loai === "duoc") {
		await ghiBot({ zaloBotName: ket.tenBot, zaloPolledAt: now });
		return true;
	}
	console.error(`[zalo] getMe ${ket.loi.lop}, ${ket.loi.ketLuan}`);
	await ghiAnToan(ghi, "kiem-bot", token, ket.loi);
	if (ket.loi.ketLuan === "token-sai") {
		await ghiBot({ zaloBotName: null, zaloWebhookUrl: null });
	}
	return false;
}
```
